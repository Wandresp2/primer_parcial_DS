PATRON_J = re.compile(r"\d+(/\d+)?")


def parsear_J(v):
    # Solo convierte lo que es inequivocamente un entero o una fraccion simple.
    # '---', '0,1,2', '1/2 or 3/2' -> no resuelto (NaN)
    if pd.isna(v):
        return np.nan
    s = str(v).strip()
    if not PATRON_J.fullmatch(s):
        return np.nan
    if "/" in s:
        a, b = s.split("/")
        return float(a) / float(b)
    return float(s)
